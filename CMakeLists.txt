cmake_minimum_required(VERSION 3.16)
project(educational_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(lab src/core.cpp)
target_include_directories(lab PUBLIC include)
target_compile_options(lab PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
target_sources(lab PRIVATE src/analysis.cpp)
add_executable(experiment_cli src/main.cpp)
target_link_libraries(experiment_cli PRIVATE lab)
add_executable(reference_tests tests/reference.cpp)
target_link_libraries(reference_tests PRIVATE lab)
add_test(NAME reference COMMAND reference_tests)
add_executable(analysis_tests tests/analysis.cpp)
target_link_libraries(analysis_tests PRIVATE lab)
add_test(NAME analysis COMMAND analysis_tests)
add_test(NAME cli_invalid COMMAND experiment_cli)
set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
add_executable(demo examples/demo.cpp)
target_link_libraries(demo PRIVATE lab)
add_executable(benchmark benchmarks/benchmark.cpp)
target_link_libraries(benchmark PRIVATE lab)
add_test(NAME cli_example COMMAND experiment_cli ${CMAKE_CURRENT_SOURCE_DIR}/examples/encoders.csv 0.0762 0.55 2048)
set_tests_properties(cli_example PROPERTIES PASS_REGULAR_EXPRESSION "step,x,y,heading")
